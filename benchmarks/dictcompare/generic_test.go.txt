package main

import "testing"

var sink int64

func run(b *testing.B, name string, f func() int64) {
	b.Run(name, func(b *testing.B) {
		for b.Loop() {
			sink = f()
		}
	})
}

func TestSaturatedMethodsDoNotAllocate(t *testing.T) {
	d := V_smallDict(100)
	for name, f := range map[string]func() int64{
		"Dict.get":                func() int64 { return V_dictGet(d, 100) },
		"conditional dictionary":  func() int64 { return V_constructOnly(100, 0) },
	} {
		if allocs := testing.AllocsPerRun(10, func() { sink = f() }); allocs != 0 {
			t.Errorf("%s: %v allocations", name, allocs)
		}
	}
}

func BenchmarkGeneric(b *testing.B) {
	d := V_smallDict(1000)
	vs := V_values(500)
	run(b, "dict-insert-1000", func() int64 { return V_dictInsert(1000) })
	run(b, "dict-get-1000", func() int64 { return V_dictGet(d, 1000) })
	run(b, "sort-500", func() int64 { return V_sortInts(vs) })
	run(b, "same-list-maybe-1000", func() int64 { return V_countSame(1000, 0) })
	run(b, "generic-list-1000", func() int64 { return V_genericList(1000) })
	run(b, "construct-maybe-1000", func() int64 { return V_constructOnly(1000, 0) })
	run(b, "show-1000", func() int64 { return V_showAll(1000, 0) })
}
