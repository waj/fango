package main

import "testing"

var hiddenCompilerSum int64
var hiddenCompilerPackage T_Source[int64]

func TestKnownPackageAllocationSlope(t *testing.T) {
 for _, n := range []int64{0, 1, 10000} {
  for _, f := range []func(int64) int64{V_explicit, V_hidden, V_knownCall, func(n int64)int64{return V_dynamic(true,n)},func(n int64)int64{return V_dynamic(false,n)}} {
   allocations := testing.AllocsPerRun(20, func(){ hiddenCompilerSum = f(n) })
   if want := n*(n+1)/2; hiddenCompilerSum != want || allocations != 0 {
    t.Fatalf("n=%d checksum=%d allocations=%g", n, hiddenCompilerSum, allocations)
   }
  }
 }
}

func TestRepackingPreservesRepresentation(t *testing.T) {
 original := V_make(10)
 allocations := testing.AllocsPerRun(20, func(){
  next := original
  for i:=0; i<1000; i++ { next = V_repack[int64](nil, next) }
  hiddenCompilerPackage = next
 })
 if allocations != 0 { t.Fatalf("repacking allocated %g objects", allocations) }
 if V_consume(original) != 55 || V_consume(hiddenCompilerPackage) != 55 { t.Fatal("repacking changed a value or its alias") }
}
