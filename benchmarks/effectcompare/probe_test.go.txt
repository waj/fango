package main

import (
	"os"
	"strconv"
	"testing"
)

var effectSink int64

func TestEffectChecks(t *testing.T) {
	for _, probe := range []struct {
		name string
		run  func() int64
		want int64
	}{
		{"state", func() int64 { return V_state(100) }, 100},
		{"cleanup", func() int64 { return V_bracket(100) }, 100},
		{"console", func() int64 { return V_console(100) }, 100},
		{"async", func() int64 { return V_async(16) }, 136},
		{"resumptive normal", func() int64 { return V_resumptiveAbort(false, 8) }, 0},
		{"resumptive abort", func() int64 { return V_resumptiveAbort(true, 8) }, 1},
	} {
		if got := probe.run(); got != probe.want {
			t.Fatalf("%s: %d != %d", probe.name, got, probe.want)
		}
	}
	for _, size := range []int64{1, 1000} {
		values := V_input(size)
		if n := testing.AllocsPerRun(20, func() { effectSink = V_fold(values) }); n != 0 {
			t.Fatalf("fold(%d): %g allocations", size, n)
		}
		if effectSink != size*(size+1)/2 {
			t.Fatal("fold checksum")
		}
	}
	data, err := os.ReadFile(os.Getenv("FANGO_EFFECT_JSON"))
	if err != nil {
		t.Fatal(err)
	}
	want, err := strconv.ParseInt(os.Getenv("FANGO_EFFECT_RECORDS"), 10, 64)
	if err != nil {
		t.Fatal(err)
	}
	if got := V_json(string(data)); got != want {
		t.Fatalf("json: %d != %d", got, want)
	}
}

func BenchmarkState(b *testing.B) {
	for b.Loop() {
		effectSink = V_state(10000)
	}
}
func BenchmarkCleanup(b *testing.B) {
	for b.Loop() {
		effectSink = V_bracket(10000)
	}
}
func BenchmarkConsole(b *testing.B) {
	for b.Loop() {
		effectSink = V_console(1000)
	}
}
func BenchmarkAsync(b *testing.B) {
	for b.Loop() {
		effectSink = V_async(64)
	}
}
func BenchmarkResumptiveNormal(b *testing.B) {
	for b.Loop() {
		effectSink = V_resumptiveAbort(false, 8)
	}
}
func BenchmarkResumptiveAbort(b *testing.B) {
	for b.Loop() {
		effectSink = V_resumptiveAbort(true, 8)
	}
}
func BenchmarkFold(b *testing.B) {
	values := V_input(10000)
	b.ResetTimer()
	for b.Loop() {
		effectSink = V_fold(values)
	}
}
func BenchmarkJSON(b *testing.B) {
	data, err := os.ReadFile(os.Getenv("FANGO_EFFECT_JSON"))
	if err != nil {
		b.Fatal(err)
	}
	input := string(data)
	b.SetBytes(int64(len(data)))
	b.ResetTimer()
	for b.Loop() {
		effectSink = V_json(input)
	}
}
